// maxBodyBytes caps request bodies; a valid request is well under 1 KB.
const maxBodyBytes = 1 << 10

var errBadRequestBody = &calc.Error{Code: calc.CodeInvalidInput, Message: "Invalid request body"}

// calculate returns the handler for one operator endpoint.
func calculate(op calc.Operator) gin.HandlerFunc {
        return func(c *gin.Context) {
                var req CalculateRequest
                c.Request.Body = http.MaxBytesReader(c.Writer, c.Request.Body, maxBodyBytes)
                if err := json.NewDecoder(c.Request.Body).Decode(&req); err != nil {
                        respondError(c, nil, errBadRequestBody)
                        return
                }

                left, err := calc.ParseNumber(req.Left)
                if err != nil {
                        respondError(c, nil, err)
                        return
                }
                right, err := calc.ParseNumber(req.Right)
                if err != nil {
                        respondError(c, nil, err)
                        return
                }

                operation := &Operation{
                        Operator: op,
                        Left:     calc.FormatNumber(left),
                        Right:    calc.FormatNumber(right),
                }
                value, err := calc.Evaluate(op, left, right)
                if err != nil {
                        respondError(c, operation, err)
                        return
                }

                result := calc.FormatNumber(value)
                c.JSON(http.StatusOK, CalculateResponse{OK: true, Operation: operation, Result: &result})
        }
}

// respondError writes a failed CalculateResponse. Problems with the request
// itself are 400s; valid requests describing an invalid operation are 422s.
func respondError(c *gin.Context, operation *Operation, err error) {
        var calcErr *calc.Error
        if !errors.As(err, &calcErr) {
                c.JSON(http.StatusInternalServerError, CalculateResponse{
                        Error: &ErrorBody{Code: "INTERNAL_ERROR", Message: "Internal error"},
                })
                return
        }

        status := http.StatusUnprocessableEntity
        if calcErr.Code == calc.CodeInvalidInput || calcErr.Code == calc.CodeOperandOutOfRange {
                status = http.StatusBadRequest
        }
        c.JSON(status, CalculateResponse{
                Operation: operation,
                Error:     &ErrorBody{Code: calcErr.Code, Message: calcErr.Message},
        })
}

func health(c *gin.Context) {
        c.JSON(http.StatusOK, gin.H{"status": "ok"})
}

// Config configures the HTTP router.
type Config struct {
        // AllowedOrigins lists origins allowed to call the API cross-origin.
        // Empty means same-origin only (e.g. through the Vite dev proxy).
        AllowedOrigins []string
        // StaticDir, if set, is a directory of frontend files (the Vite build)
        // served for every non-API path.
        StaticDir string
}

// NewRouter builds the router: POST /api/<operator> for every operator,
// GET /api/health, and the frontend files when cfg.StaticDir is set.
func NewRouter(cfg Config) *gin.Engine {
        r := gin.New()
        r.Use(gin.Logger(), gin.Recovery(), cors(cfg.AllowedOrigins))
        // We don't sit behind a proxy, so don't trust X-Forwarded-For.
        _ = r.SetTrustedProxies(nil)

        api := r.Group("/api")
        api.GET("/health", health)
        for _, op := range calc.Operators {
                api.POST("/"+string(op), calculate(op))
        }
        if cfg.StaticDir != "" {
                r.NoRoute(staticFiles(cfg.StaticDir))
        }
        return r
}

// cors allows cross-origin requests from the given origins and answers
// preflight requests.
func cors(allowedOrigins []string) gin.HandlerFunc {
        return func(c *gin.Context) {
                origin := c.GetHeader("Origin")
                if origin != "" && slices.Contains(allowedOrigins, origin) {
                        h := c.Writer.Header()
                        h.Set("Access-Control-Allow-Origin", origin)
                        h.Set("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
                        h.Set("Access-Control-Allow-Headers", "Content-Type")
                        h.Add("Vary", "Origin")
                        if c.Request.Method == http.MethodOptions {
                                c.AbortWithStatus(http.StatusNoContent)
                                return
                        }
                }
                c.Next()
        }
}

// staticFiles serves the built frontend from dir for any GET/HEAD request
// that no API route matched. Everything else gets a 404.
//
// Vite puts content-hashed files under /assets/, so those can be cached
// forever. Other files (index.html) must be revalidated so a new deploy is
// picked up.
func staticFiles(dir string) gin.HandlerFunc {
        files := http.FileServer(http.Dir(dir))
        return func(c *gin.Context) {
                path := c.Request.URL.Path
                method := c.Request.Method
                isAPI := path == "/api" || strings.HasPrefix(path, "/api/")
                // No directory listings: only "/" may end in a slash (it serves index.html).
                isDir := path != "/" && strings.HasSuffix(path, "/")
                if (method != http.MethodGet && method != http.MethodHead) || isAPI || isDir {
                        c.Status(http.StatusNotFound)
                        return
                }

                if strings.HasPrefix(path, "/assets/") {
                        c.Header("Cache-Control", "public, max-age=31536000, immutable")
                } else {
                        c.Header("Cache-Control", "no-cache")
                }
                files.ServeHTTP(c.Writer, c.Request)
        }
}

// Code is a stable, machine-readable identifier for a calculation error.
type Code string

const (
        CodeDivisionByZero      Code = "DIVISION_BY_ZERO"
        CodeZeroToNegativePower Code = "ZERO_TO_NEGATIVE_POWER"
        CodeComplexResult       Code = "COMPLEX_RESULT"
        CodeResultTooLarge      Code = "RESULT_TOO_LARGE"
        CodeResultTooSmall      Code = "RESULT_TOO_SMALL"
        CodeOperandOutOfRange   Code = "OPERAND_OUT_OF_RANGE"
        CodeInvalidInput        Code = "INVALID_INPUT"
)

// Error is a calculation error with a stable code and a human-readable message.
type Error struct {
        Code    Code
        Message string
}

func (e *Error) Error() string { return e.Message }

// Sentinel errors returned by this package. Compare with errors.Is.
var (
        ErrDivisionByZero      = &Error{CodeDivisionByZero, "Division by 0"}
        ErrZeroToNegativePower = &Error{CodeZeroToNegativePower, "0 raised to a negative power"}
        ErrComplexResult       = &Error{CodeComplexResult, "Result is not a real number"}
        ErrResultTooLarge      = &Error{CodeResultTooLarge, "Result is too large"}
        ErrResultTooSmall      = &Error{CodeResultTooSmall, "Result is too small"}
        ErrOperandOutOfRange   = &Error{CodeOperandOutOfRange, "Number is out of range"}
        ErrInvalidInput        = &Error{CodeInvalidInput, "Invalid number"}
        ErrUnknownOperator     = &Error{CodeInvalidInput, "Unknown operator"}
)

package calc

import "math"

// Evaluate applies op to a and b. The result is rounded to SignificantDigits
// and checked against the limits. Errors are always *Error values.
func Evaluate(op Operator, a, b float64) (float64, error) {
        if !op.Valid() {
                return 0, ErrUnknownOperator
        }
        if err := checkOperand(a); err != nil {
                return 0, err
        }
        if err := checkOperand(b); err != nil {
                return 0, err
        }

        var r float64
        // mustBeNonZero is set when the exact result can't be 0, so a 0 from the
        // float operation means it underflowed.
        var mustBeNonZero bool

        switch op {
        case Add:
                r = a + b
        case Subtract:
                r = a - b
        case Multiply:
                r = a * b
                mustBeNonZero = a != 0 && b != 0
        case Divide:
                if b == 0 {
                        return 0, ErrDivisionByZero
                }
                r = a / b
                mustBeNonZero = a != 0
        case Power:
                if a == 0 && b < 0 {
                        return 0, ErrZeroToNegativePower
                }
                if a < 0 && b != math.Trunc(b) {
                        return 0, ErrComplexResult
                }
                r = math.Pow(a, b)
                mustBeNonZero = a != 0
        }

        if math.IsInf(r, 0) {
                return 0, ErrResultTooLarge
        }
        if r == 0 && mustBeNonZero {
                return 0, ErrResultTooSmall
        }
        r = roundSignificant(r)
        if err := checkResult(r); err != nil {
                return 0, err
        }
        return normalizeZero(r), nil
}

package calc

import "math"

const (
        // MaxMagnitude is the largest absolute value accepted as an operand or
        // returned as a result (inclusive).
        MaxMagnitude = 1e100
        // MinMagnitude is the smallest non-zero absolute value accepted as an
        // operand or returned as a result (inclusive).
        MinMagnitude = 1e-100
        // SignificantDigits is the precision results are rounded to. It sits just
        // below float64's ~15.9 decimal digits, which hides binary artifacts such
        // as 0.1 + 0.2 = 0.30000000000000004.
        SignificantDigits = 15
)

// inRange reports whether x is zero or within [MinMagnitude, MaxMagnitude].
func inRange(x float64) bool {
        abs := math.Abs(x)
        return x == 0 || (abs >= MinMagnitude && abs <= MaxMagnitude)
}

// checkOperand validates a number used as an operand.
func checkOperand(x float64) error {
        if math.IsNaN(x) || math.IsInf(x, 0) {
                return ErrInvalidInput
        }
        if !inRange(x) {
                return ErrOperandOutOfRange
        }
        return nil
}

// checkResult validates an (already rounded) result.
func checkResult(x float64) error {
        switch {
        case math.IsNaN(x):
                return ErrComplexResult
        case math.Abs(x) > MaxMagnitude:
                return ErrResultTooLarge
        case !inRange(x):
                return ErrResultTooSmall
        }
        return nil
}

import (
        "errors"
        "math"
        "regexp"
        "strconv"
        "strings"
)

// numberPattern accepts plain decimal numbers with an optional sign and
// exponent: "12", "-0.5", "5.", ".5", "3e+12", "1.5E-7". It deliberately
// rejects forms strconv.ParseFloat would otherwise accept, such as "Inf",
// "NaN", hex floats and underscores.
var numberPattern = regexp.MustCompile(`^[+-]?(?:\d+\.?\d*|\.\d+)(?:[eE][+-]?\d+)?$`)

// ParseNumber parses an operand string and checks it against the limits.
func ParseNumber(s string) (float64, error) {
        if !numberPattern.MatchString(s) {
                return 0, ErrInvalidInput
        }
        x, err := strconv.ParseFloat(s, 64)
        if errors.Is(err, strconv.ErrRange) {
                return 0, ErrOperandOutOfRange
        }
        if err != nil {
                return 0, ErrInvalidInput
        }
        // ParseFloat silently underflows tiny values (e.g. "1e-400") to 0.
        if x == 0 && hasNonZeroDigit(s) {
                return 0, ErrOperandOutOfRange
        }
        if err := checkOperand(x); err != nil {
                return 0, err
        }
        return normalizeZero(x), nil
}

// hasNonZeroDigit reports whether the mantissa of a number string (the part
// before any exponent) contains a digit other than 0.
func hasNonZeroDigit(s string) bool {
        mantissa, _, _ := strings.Cut(strings.ToLower(s), "e")
        return strings.ContainsAny(mantissa, "123456789")
}

// FormatNumber returns the canonical string for x: the shortest representation
// that parses back to the same float64. Like JavaScript's Number#toString, it
// uses plain notation for 1e-6 <= |x| < 1e21 and exponent notation otherwise.
// This is a machine-readable format, not display text.
func FormatNumber(x float64) string {
        x = normalizeZero(x)
        abs := math.Abs(x)
        format := byte('f')
        if abs != 0 && (abs < 1e-6 || abs >= 1e21) {
                format = 'e'
        }
        s := strconv.FormatFloat(x, format, -1, 64)
        if format == 'e' {
                // Drop exponent zero-padding: "1e-07" -> "1e-7".
                if n := len(s); n >= 4 && s[n-4] == 'e' && s[n-2] == '0' {
                        s = s[:n-2] + s[n-1:]
                }
        }
        return s
}

// roundSignificant rounds x to SignificantDigits significant digits.
func roundSignificant(x float64) float64 {
        if x == 0 || math.IsNaN(x) || math.IsInf(x, 0) {
                return x
        }
        r, _ := strconv.ParseFloat(strconv.FormatFloat(x, 'e', SignificantDigits-1, 64), 64)
        return r
}

// normalizeZero turns -0 into 0.
func normalizeZero(x float64) float64 {
        if x == 0 {
                return 0
        }
        return x
}

// Package calc holds the calculator's domain logic: parsing operands,
// evaluating binary operations, enforcing limits and producing canonical
// number strings. It has no knowledge of HTTP or of how numbers are displayed.
package calc

// Operator identifies a binary operation. Values are stable identifiers meant
// to be exposed to clients, not display symbols.
type Operator string

const (
        Add      Operator = "add"
        Subtract Operator = "subtract"
        Multiply Operator = "multiply"
        Divide   Operator = "divide"
        Power    Operator = "power"
)

// Operators lists every supported operator.
var Operators = []Operator{Add, Subtract, Multiply, Divide, Power}

// Valid reports whether op is a supported operator.
func (op Operator) Valid() bool {
        switch op {
        case Add, Subtract, Multiply, Divide, Power:
                return true
        }
        return false
}

package main

import (
        "log"
        "net"
        "os"
        "strings"

        "calculator/internal/api"
)

// Configuration comes from the environment:
//
//        HOST             address to bind (default "localhost")
//        PORT             port to listen on (default "8080")
//        ALLOWED_ORIGINS  comma-separated origins allowed cross-origin (default none)
//        STATIC_DIR       directory of built frontend files to serve (default none: API only)
func main() {
        addr := net.JoinHostPort(getenv("HOST", "localhost"), getenv("PORT", "8080"))
        router := api.NewRouter(api.Config{
                AllowedOrigins: splitList(os.Getenv("ALLOWED_ORIGINS")),
                StaticDir:      os.Getenv("STATIC_DIR"),
        })

        log.Printf("calculator listening on %s", addr)
        if err := router.Run(addr); err != nil {
                log.Fatal(err)
        }
}

func getenv(key, fallback string) string {
        if v := os.Getenv(key); v != "" {
                return v
        }
        return fallback
}

func splitList(s string) []string {
        var out []string
        for _, item := range strings.Split(s, ",") {
                if item = strings.TrimSpace(item); item != "" {
                        out = append(out, item)
                }
        }
        return out
}
